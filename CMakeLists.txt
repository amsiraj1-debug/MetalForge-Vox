cmake_minimum_required(VERSION 3.24)
project(VocalMorph VERSION 0.1.0 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
option(VM_BUILD_PLUGIN "Build Windows VST3" ${WIN32})
include(CTest)
add_subdirectory(core)
if(VM_BUILD_PLUGIN)
  if(NOT WIN32)
    message(FATAL_ERROR "The VocalMorph plugin supports Windows x64 only")
  endif()
  include(FetchContent)
  FetchContent_Declare(JUCE GIT_REPOSITORY https://github.com/juce-framework/JUCE.git GIT_TAG 29396c22c93392d6738e021b83196283d6e4d850 GIT_SHALLOW FALSE)
  FetchContent_MakeAvailable(JUCE)
  add_subdirectory(plugin)
endif()
if(BUILD_TESTING)
  add_executable(core_tests tests/core_tests.cpp)
  target_link_libraries(core_tests PRIVATE VocalMorphCore)
  if(NOT WIN32)
    target_link_libraries(core_tests PRIVATE pthread)
  endif()
  add_test(NAME core COMMAND core_tests)
endif()
install(FILES README.md LICENSE docs/INSTALL.md docs/MODELS.md DESTINATION .)
set(CPACK_GENERATOR ZIP)
set(CPACK_PACKAGE_FILE_NAME "VocalMorph-${PROJECT_VERSION}-Windows-x64")
include(CPack)
